cmake_minimum_required(VERSION 3.22)
project(WoofStore VERSION 1.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.6 REQUIRED COMPONENTS Widgets Network Concurrent DBus)
find_package(ZLIB REQUIRED)
find_package(KF6CoreAddons REQUIRED)
find_package(KF6Parts REQUIRED)
find_package(AppStreamQt REQUIRED)
add_library(store-core backend.cpp catalog.cpp merge.cpp review.cpp leftovers.cpp)
# Lets a development build read metadata/ from the source tree without installing.
target_compile_definitions(store-core PRIVATE WOOFSTORE_SOURCE_METADATA="${CMAKE_CURRENT_SOURCE_DIR}/metadata")
target_link_libraries(store-core PUBLIC Qt6::Widgets Qt6::Network Qt6::Concurrent AppStreamQt ZLIB::ZLIB)
target_compile_options(store-core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(woof-store main.cpp)
# Checks and screenshot previews read the small test catalogue from the source tree.
target_compile_definitions(woof-store PRIVATE WOOFSTORE_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
target_link_libraries(woof-store PRIVATE store-core Qt6::DBus KF6::CoreAddons KF6::Parts)
target_compile_options(woof-store PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(store-tests tests.cpp)
  target_link_libraries(store-tests PRIVATE store-core)
  add_test(NAME store-checks COMMAND store-tests)
  add_test(NAME explore-check COMMAND woof-store --explore-check)
  set_tests_properties(explore-check PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_test(NAME prototype-check COMMAND woof-store --prototype-check)
  set_tests_properties(prototype-check PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_test(NAME sidebar-check COMMAND woof-store --sidebar-check)
  set_tests_properties(sidebar-check PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  set_tests_properties(store-checks PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
install(TARGETS woof-store RUNTIME DESTINATION bin)
install(FILES io.github.laskywolf.WoofStore.desktop DESTINATION share/applications)
install(FILES icons/io.github.laskywolf.WoofStore.svg DESTINATION share/icons/hicolor/scalable/apps)
install(DIRECTORY metadata/ DESTINATION share/woofstore/metadata)
