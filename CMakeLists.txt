cmake_minimum_required(VERSION 3.22)
project(CachyStore VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.6 REQUIRED COMPONENTS Widgets Network)
find_package(KF6CoreAddons REQUIRED)
find_package(KF6Parts REQUIRED)
find_package(AppStreamQt REQUIRED)
add_library(store-core backend.cpp)
target_link_libraries(store-core PUBLIC Qt6::Widgets Qt6::Network AppStreamQt)
target_compile_options(store-core PRIVATE -Wall -Wextra -Wpedantic)
qt_add_resources(explore-resources explore.qrc)
add_executable(cachy-store main.cpp ${explore-resources})
target_link_libraries(cachy-store PRIVATE store-core KF6::CoreAddons KF6::Parts)
target_compile_options(cachy-store PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(store-tests tests.cpp)
  target_link_libraries(store-tests PRIVATE store-core)
  add_test(NAME store-checks COMMAND store-tests)
  add_test(NAME explore-check COMMAND cachy-store --explore-check)
  set_tests_properties(explore-check PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_test(NAME prototype-check COMMAND cachy-store --prototype-check)
  set_tests_properties(prototype-check PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_test(NAME sidebar-check COMMAND cachy-store --sidebar-check)
  set_tests_properties(sidebar-check PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  set_tests_properties(store-checks PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
install(TARGETS cachy-store RUNTIME DESTINATION bin)
install(FILES org.local.CachyStore.desktop DESTINATION share/applications)
